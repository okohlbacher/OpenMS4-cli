cmake_minimum_required(VERSION 3.24)
project(OpenMSCLI VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(GenerateExportHeader)
include(CTest)
include(cmake/OpenMS4Dependencies.cmake)
if(BUILD_TESTING)
  openms4_find_core(COMPONENTS TestSupport)
else()
  openms4_find_core()
endif()
openms4_source_revision(OpenMSCLI_SOURCE_REVISION OpenMSCLI_SOURCE_DIRTY)
file(GLOB cli_sources CONFIGURE_DEPENDS source/APPLICATIONS/*.cpp)
add_library(OpenMS_CLI SHARED ${cli_sources})
add_library(OpenMS::CLI ALIAS OpenMS_CLI)
openms4_install_rpath(OpenMS_CLI)
set_target_properties(OpenMS_CLI PROPERTIES EXPORT_NAME CLI VERSION ${PROJECT_VERSION} SOVERSION 1)
target_compile_features(OpenMS_CLI PUBLIC cxx_std_23)
target_link_libraries(OpenMS_CLI PUBLIC OpenMS::Core PRIVATE ${CMAKE_DL_LIBS})
generate_export_header(OpenMS_CLI EXPORT_FILE_NAME generated/OpenMS/APPLICATIONS/OpenMSCLIConfig.h EXPORT_MACRO_NAME OPENMSCLI_DLLAPI)
target_include_directories(OpenMS_CLI PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
install(TARGETS OpenMS_CLI EXPORT OpenMSCLITargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
# The generated test_config.h contains this checkout's fixture paths and is not SDK API.
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/generated/OpenMS/APPLICATIONS/OpenMSCLIConfig.h
  DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/OpenMS/APPLICATIONS)
set(OpenMSCLI_VERSION ${PROJECT_VERSION})
configure_package_config_file(cmake/OpenMSCLIConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/OpenMSCLIConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OpenMSCLI)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/OpenMSCLIConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(EXPORT OpenMSCLITargets NAMESPACE OpenMS:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OpenMSCLI)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/OpenMSCLIConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/OpenMSCLIConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OpenMSCLI)
if(BUILD_TESTING)
  set(CF_OPENMS_TEST_DATA_PATH "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/")
  configure_file(tests/test_config.h.in generated/OpenMS/test_config.h @ONLY)
  file(GLOB tests CONFIGURE_DEPENDS tests/source/*_test.cpp)
  foreach(source IN LISTS tests)
    get_filename_component(name ${source} NAME_WE)
    add_executable(${name} ${source} ${OpenMS_TEST_SUPPORT_SOURCE})
    target_link_libraries(${name} PRIVATE OpenMS::CLI OpenMS::TestFramework)
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "OPENMS_TOOL_PREFIX_PATH=${CMAKE_CURRENT_SOURCE_DIR}/tests/registry;OPENMS_DISABLE_UPDATE_CHECK=ON")
  endforeach()
endif()
